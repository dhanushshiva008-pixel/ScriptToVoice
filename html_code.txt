<script>
async function generateVoice(){
  const script=document.getElementById('script').value.trim();
  const voice=document.getElementById('voice').value;
  const narration=document.getElementById('narration').value;
  const status=document.getElementById('status');
  const button=document.getElementById('generate');
  const player=document.getElementById('player');
  const download=document.getElementById('download');
  if(!script){status.textContent='ముందుగా మీ స్క్రిప్ట్ టైప్ చేయండి.';return;}
  button.disabled=true;
  status.textContent='వాయిస్ తయారవుతోంది. దయచేసి వేచి ఉండండి...';
  player.hidden=true;
  download.hidden=true;
  try{
    const response=await fetch('/generate',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({script:script,voice:voice,narration:narration})
    });
    const data=await response.json();
    if(!response.ok) throw new Error(data.error||'వాయిస్ తయారు కాలేదు.');
    player.src=data.url;
    player.hidden=false;
    download.href=data.url;
    download.hidden=false;
    status.textContent='వాయిస్ తయారైంది! వినండి లేదా MP3 డౌన్‌లోడ్ చేసుకోండి.';
  }catch(error){
    status.textContent='సమస్య: '+error.message;
  }finally{
    button.disabled=false;
  }
}
</script>
